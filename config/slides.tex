\setlength{\leftmargini}{1.2em}

\newcounter{secaoComSlides}
\newcounter{primeiroFrameDaSecao}

\newcommand{\RegistrarTotalSecaoAux}[2]{%
  \expandafter\gdef\csname totalDaSecao@#1\endcsname{#2}%
}

\makeatletter

\newcommand{\SalvarTotalSecao}{%
  \begingroup
    \edef\numeroDaSecao{%
      \number\value{secaoComSlides}%
    }%
    \edef\totalDaSecao{%
      \number\numexpr
        \value{framenumber}
        -\value{primeiroFrameDaSecao}+1
      \relax
    }%
    \immediate\write\@auxout{%
      \string\RegistrarTotalSecaoAux
        {\numeroDaSecao}{\totalDaSecao}%
    }%
  \endgroup
}

\makeatother

\AtBeginSection[]{%
  \ifnum\value{secaoComSlides}>0\relax
    \SalvarTotalSecao
  \fi

  \setcounter{secaoComSlides}{\value{section}}%

  \setcounter{primeiroFrameDaSecao}{%
    \numexpr\value{framenumber}+1\relax
  }%
}

\AtEndDocument{%
  \ifnum\value{secaoComSlides}>0\relax
    \SalvarTotalSecao
  \fi
}

\newcommand{\DesenharMolduraSlide}{%

  \ifnum\value{framenumber}>0\relax

    \begin{tikzpicture}[
      remember picture,
      overlay
    ]

      \coordinate (SW) at
        (current page.south west);

      \coordinate (NE) at
        (current page.north east);

      \draw[
        draw=bordaSlide,
        line width=.70pt,
        rounded corners=.36cm
      ]
        ($(SW)+(.36cm,.32cm)$)
        rectangle
        ($(NE)+(-.36cm,-.32cm)$);

        \def\TotalPontos{1}
        \def\PontoAtual{1}

        \ifnum\value{secaoComSlides}>0\relax

        \ifcsname totalDaSecao@\number\value{secaoComSlides}\endcsname
            \edef\TotalPontos{%
            \csname totalDaSecao@\number\value{secaoComSlides}\endcsname
            }
        \fi

        \edef\PontoAtual{%
            \number\numexpr
            \value{framenumber}
            -\value{primeiroFrameDaSecao}+1
            \relax
        }

        \fi

        \pgfmathtruncatemacro{\TotalPontos}{
        max(1,\TotalPontos)
        }

        \pgfmathtruncatemacro{\PontoAtual}{
        max(1,\PontoAtual)
        }

        \pgfmathsetmacro{\EspacoPontos}{
        min(.275,5.8/\TotalPontos)
        }

        \foreach \i in {1,...,\TotalPontos} {

        \pgfmathsetmacro{\Deslocamento}{
            (\i-(\TotalPontos+1)/2)*\EspacoPontos
        }

        \ifnum\i=\PontoAtual\relax

            \fill[pontoAtivo]
            ($(current page.south)+
                (\Deslocamento cm,.64cm)$)
            circle [radius=.061cm];

        \else

            \fill[pontoInativo]
            ($(current page.south)+
                (\Deslocamento cm,.64cm)$)
            circle [radius=.061cm];

        \fi
        }

      \fill[white]
        ($(current page.south)+
          (-.37cm,.22cm)$)
        rectangle
        ($(current page.south)+
          (.37cm,.42cm)$);

      \node[
        font=\fontsize{8.5}{10}\selectfont,
        text=azulPrincipal
      ]
        at ($(current page.south)+
          (0,.32cm)$)
        {
          \insertframenumber/
          \inserttotalframenumber
        };

    \end{tikzpicture}%

  \fi
}

\setbeamertemplate{background}{
  \DesenharMolduraSlide
}
