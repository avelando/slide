\cs_new_protected:Npn \__capa_marca_padrao:n #1
  {
    \begin{tikzpicture}[baseline=(marca.base)]
      \node[
        draw=azulPrincipal!30,
        fill=white,
        rounded~corners=3pt,
        text=azulPrincipal,
        inner~xsep=10pt,
        inner~ysep=5pt,
        font=\bfseries\large
      ] (marca) {#1};
    \end{tikzpicture}
  }


\cs_new_protected:Npn \__capa_instituicao:nnn #1#2#3
  {
    \begin{minipage}[c][1.18cm][c]{\l__capa_logo_width_dim}
      \centering

      \tl_if_blank:nTF {#3}
        {\__capa_marca_padrao:n {#1}}
        {
          \IfFileExists{#3}
            {
              \includegraphics[
                width=\dim_eval:n{\l__capa_logo_width_dim-0.24cm},
                height=1.05cm,
                keepaspectratio
              ]{#3}
            }
            {\__capa_marca_padrao:n {#1}}
        }
    \end{minipage}
  }


\cs_new_protected:Npn \__capa_logos:
  {
    \int_set:Nn \l__capa_total_int
      {\seq_count:N \g__capa_instituicoes_seq}

    \int_compare:nNnT {\l__capa_total_int} > {0}
      {
        \dim_set:Nn \l__capa_logo_width_dim
          {
            \dim_min:nn
              {3.25cm}
              {.81\paperwidth/\l__capa_total_int}
          }

        \hbox to .92\paperwidth
          {
            \hfill
            \seq_map_inline:Nn \g__capa_instituicoes_seq
              {##1\hfill}
          }
      }
  }


\cs_new_protected:Npn \__capa_avatar_fallback:n #1
  {
    \node[
      text=azulPrincipal,
      font=\bfseries\large
    ] at (0,0)
      {\tl_head:n {#1}};
  }


\cs_new_protected:Npn \__capa_avatar:nn #1#2
  {
    \begin{tikzpicture}[baseline=(base)]

      \coordinate (base) at
        (0,-\dim_eval:n{.5\l__capa_avatar_dim});

      \fill[azulClaro]
        (0,0) circle
        [radius=\dim_eval:n{.5\l__capa_avatar_dim}];

      \tl_if_blank:nTF {#2}
        {\__capa_avatar_fallback:n {#1}}
        {
          \IfFileExists{#2}
            {
              \hbox_set:Nn \l__capa_imagem_box
                {\includegraphics[width=1cm]{#2}}

              \begin{scope}

                \clip (0,0) circle
                  [radius=\dim_eval:n{.5\l__capa_avatar_dim}];

                \dim_compare:nNnTF
                  {\box_ht:N \l__capa_imagem_box}
                  >
                  {\box_wd:N \l__capa_imagem_box}
                  {
                    \node[
                      anchor=center,
                      inner~sep=0pt,
                      outer~sep=0pt
                    ] at (0,0)
                      {
                        \includegraphics[
                          width=\l__capa_avatar_dim
                        ]{#2}
                      };
                  }
                  {
                    \node[
                      anchor=center,
                      inner~sep=0pt,
                      outer~sep=0pt
                    ] at (0,0)
                      {
                        \includegraphics[
                          height=\l__capa_avatar_dim
                        ]{#2}
                      };
                  }

              \end{scope}
            }
            {\__capa_avatar_fallback:n {#1}}
        }

      \draw[
        azulPrincipal!30,
        line~width=.6pt
      ]
        (0,0) circle
        [radius=\dim_eval:n{.5\l__capa_avatar_dim}];

    \end{tikzpicture}
  }


\cs_new_protected:Npn \__capa_autor:nnn #1#2#3
  {
    \begin{minipage}[t]{\l__capa_autor_width_dim}
      \centering

      \__capa_avatar:nn {#1}{#3}

      \par\vspace{0.09cm}

      {
        \color{azulEscuro}
        \bfseries

        \int_compare:nNnTF
          {\seq_count:N \g__capa_autores_seq} > {4}
          {\fontsize{7.5}{8.5}\selectfont}
          {\fontsize{9}{10.5}\selectfont}

        #1\par
      }

      \vspace{0.04cm}

      {
        \color{cinzaTexto}
        \fontsize{8}{9}\selectfont
        #2\par
      }

    \end{minipage}
  }

% Distribuição dos autores por linha

\cs_new_protected:Npn \__capa_linha_autores:nn #1#2
  {
    \hbox to .94\paperwidth
      {
        \hfill

        \int_zero:N \l__capa_index_int

        \seq_map_inline:Nn \g__capa_autores_seq
          {
            \int_incr:N \l__capa_index_int

            \int_compare:nNnT
              {\l__capa_index_int} >
              {\int_eval:n{#1-1}}
              {
                \int_compare:nNnT
                  {\l__capa_index_int} <
                  {\int_eval:n{#2+1}}
                  {##1\hfill}
              }
          }
      }
  }


\cs_new_protected:Npn \__capa_bloco_autores:
  {
    \int_set:Nn \l__capa_total_int
      {\seq_count:N \g__capa_autores_seq}

    \int_set:Nn \l__capa_colunas_int
      {\int_min:nn {4}{\l__capa_total_int}}

    \int_compare:nNnT {\l__capa_colunas_int} > {0}
      {
        \dim_set:Nn \l__capa_autor_width_dim
          {
            \dim_min:nn
              {3.12cm}
              {.82\paperwidth/\l__capa_colunas_int}
          }

        \int_compare:nNnTF {\l__capa_total_int} > {4}
          {
            \dim_set:Nn \l__capa_avatar_dim {0.88cm}

            \begin{tikzpicture}[remember~picture,overlay]

              \node[anchor=north] at
                ([yshift=-4.03cm]current~page.north)
                {\__capa_linha_autores:nn {1}{4}};

              \node[anchor=north] at
                ([yshift=-6.32cm]current~page.north)
                {\__capa_linha_autores:nn {5}{8}};

            \end{tikzpicture}
          }
          {
            \dim_set:Nn \l__capa_avatar_dim {1.35cm}

            \begin{tikzpicture}[remember~picture,overlay]

              \node[anchor=north] at
                ([yshift=-4.79cm]current~page.north)
                {\__capa_linha_autores:nn {1}{4}};

            \end{tikzpicture}
          }
      }
  }
