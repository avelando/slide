\NewDocumentCommand{\FormatoCapa}{m}
  {
    \str_case:nnF {#1}
      {
        {texto}{\tl_gset:Nn \g__capa_formato_tl {texto}}
        {fotos}{\tl_gset:Nn \g__capa_formato_tl {fotos}}
      }
      {\PackageError{capa}{Formato~desconhecido:~#1}{Use~texto~ou~fotos.}}
  }

\cs_new_protected:Npn \__capa_avatar:nn #1#2
  {
    \begin{tikzpicture}[baseline=(base)]
      \coordinate (base) at (0,-\dim_eval:n {.5\l__capa_avatar_dim});
      \fill[azulClaro] (0,0) circle
        [radius=\dim_eval:n {.5\l__capa_avatar_dim}];
      \tl_if_blank:nTF {#2}

        {
          \IfFileExists{#2}
            {
              \hbox_set:Nn \l__capa_imagem_box
                {\includegraphics[width=1cm]{#2}}
              \begin{scope}
                \clip (0,0) circle
                  [radius=\dim_eval:n {.5\l__capa_avatar_dim}];
                \dim_compare:nNnTF
                  {\box_ht:N \l__capa_imagem_box} > {\box_wd:N \l__capa_imagem_box}
                  {
                    \node[anchor=center,inner~sep=0pt,outer~sep=0pt] at (0,0)
                      {\includegraphics[width=\l__capa_avatar_dim]{#2}};
                  }
                  {
                    \node[anchor=center,inner~sep=0pt,outer~sep=0pt] at (0,0)
                      {\includegraphics[height=\l__capa_avatar_dim]{#2}};
                  }
              \end{scope}
            }
            {\__capa_avatar_fallback:n {#1}}
        }
      \draw[azulPrincipal!30,line~width=.6pt] (0,0) circle
        [radius=\dim_eval:n {.5\l__capa_avatar_dim}];
    \end{tikzpicture}
  }

\NewDocumentCommand{\Capa}{}
  {
    \int_set:Nn \l__capa_numero_autores_int
      {\seq_count:N \g__capa_autores_seq}
    \int_compare:nNnT {\l__capa_numero_autores_int} = {0}
      {\PackageError{capa}{Nenhum~autor~cadastrado}{Adicione~um~autor~em~dados.tex.}}
    \tl_if_eq:NnT \g__capa_formato_tl {texto}
      {
        \int_compare:nNnT {\l__capa_numero_autores_int} > {9}
          {\PackageError{capa}{O~formato~texto~aceita~no~maximo~9~autores}{Divida~a~capa~ou~adapte~o~layout.}}
      }
    \tl_if_eq:NnT \g__capa_formato_tl {fotos}
      {
        \int_compare:nNnT {\l__capa_numero_autores_int} > {5}
          {\PackageError{capa}{O~formato~fotos~aceita~no~maximo~5~autores}{Use~FormatoCapa{texto}.}}
      }
    \begin{frame}[plain]
      \begin{tikzpicture}[remember~picture,overlay]
        \fill[fundoCapa]
          (current~page.south~west) rectangle (current~page.north~east);
        \fill[azulClaro!78]
          ([xshift=-.20cm,yshift=-.26cm]current~page.south~east)
          circle (1.55cm);
        \fill[azulClaro!50]
          ([xshift=-.05cm,yshift=-.34cm]current~page.north~west)
          circle (.8cm);
        \fill[azulPrincipal]
          (current~page.south~west) rectangle
          ([yshift=.14cm]current~page.south~east);
        \node[anchor=north] at
          ([yshift=-.44cm]current~page.north)
          {\__capa_logos:};
        \node[anchor=north,text=azulEscuro] at
          ([yshift=-2.25cm]current~page.north)
          {
            \parbox{.86\paperwidth}
              {
                \centering\fontsize{13.5}{17.5}\selectfont\bfseries
                \tl_use:N \g__capa_titulo_tl\par
                \tl_if_blank:VTF \g__capa_subtitulo_tl
                  {}
                  {
                    \vspace{.12cm}
                    {\color{azulPrincipal}\fontsize{9}{11}\selectfont
                     \normalfont\tl_use:N \g__capa_subtitulo_tl\par}
                  }
              }
          };
      \end{tikzpicture}
      \tl_if_eq:NnTF \g__capa_formato_tl {texto}
        {\__capa_autores_texto:}
        {\__capa_autores_fotos:}
    \end{frame}
  }
