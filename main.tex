\section{Introdução}

\begin{frame}[t]{Introdução}

  \EspacoAposTitulo

  \begin{itemize}
    \setlength{\itemsep}{\EspacoEntreTopicos}

    \item \textbf{Contexto:}
    redes complexas representam sistemas de entidades interconectadas
    (relações sociais, colaborações científicas, comunicações e
    infraestruturas físicas).

    \item \textbf{Predição de links:}
    estimar conexões ausentes ou futuras a partir da topologia observada
    — tarefa central em redes sociais, sistemas de recomendação e
    recuperação de informação.

    \item \textbf{Problema:}
    há muitos índices topológicos (locais e quase-locais), mas seus
    resultados costumam ser reportados sob condições heterogêneas,
    dificultando a comparação.

    \item \textbf{Objetivo:}
    avaliar empiricamente \dest{19 métodos topológicos} de predição de
    links em \dest{8 redes reais} de 4 domínios, sob um
    \dest{protocolo único e uniforme}.

  \end{itemize}

\end{frame}

\begin{frame}[t]{Questões de pesquisa e contribuições}

  \EspacoAposTitulo

  \begin{itemize}
    \setlength{\itemsep}{\EspacoEntreTopicos}

    \item \textbf{RQ1:} o desempenho dos métodos varia entre redes de
    domínios diferentes?
    \item \textbf{RQ2:} métodos quase-locais superam os métodos locais
    simples?
    \item \textbf{RQ3:} existe um método dominante entre os avaliados?
    \item \textbf{RQ4:} propriedades estruturais ajudam a explicar a
    dificuldade da tarefa?

  \end{itemize}

  \vspace{0.30cm}
  \begin{block}{Contribuição principal}
    Não propor um novo índice, mas uma \dest{análise comparativa}
    do comportamento de métodos topológicos sob um protocolo comum,
    incluindo ranqueamento, significância estatística e a relação
    (exploratória) entre estrutura e dificuldade.
  \end{block}

\end{frame}

\section{Metodologia experimental}

\begin{frame}[t]{Protocolo de avaliação}

  \EspacoAposTitulo

  \begin{itemize}
    \setlength{\itemsep}{\EspacoEntreTopicos}

    \item \textbf{Tarefa:} remover arestas observadas, pontuar pares
    candidatos no grafo de treino e verificar se as arestas removidas
    são ranqueadas acima de não-arestas amostradas.

    \item \textbf{Validação cruzada:} \dest{5 folds}; uma árvore geradora
    é mantida fixa e apenas as arestas restantes são distribuídas,
    preservando a conectividade do grafo de treino.

    \item \textbf{Amostragem negativa balanceada:} um par negativo para
    cada par positivo (razão 1:1) — comparação controlada, não a
    imbalanceada natural.

    \item \textbf{Métrica principal:} \dest{Average Precision (AP)},
    pois a predição é tratada como tarefa de ranqueamento.
    Complementares: ROC-AUC, Precision, NDCG (em $K=|E_i^+|$).
  \end{itemize}

\end{frame}

\begin{frame}[t]{Significância estatística do ranqueamento}

  \EspacoAposTitulo

  \begin{itemize}
    \setlength{\itemsep}{\EspacoEntreTopicos}

    \item \textbf{Teste de Friedman} sobre as 8 redes rejeita a hipótese
    de desempenho igual (estatística $=20{,}55$, $p<0{,}001$).

    \item \textbf{Wilcoxon pareado (correção de Holm)} entre os 4
    melhores (SRW-l3, LPI-beta-0.001, RA-CNI, PFP-l3):
    \begin{itemize}
      \setlength{\itemsep}{0.10cm}
      \item SRW-l3 \dest{superior} a PFP-l3 e RA-CNI ($p_{adj}=0{,}047$);
      o mesmo vale para LPI-beta-0.001.
      \item SRW-l3 vs.\ LPI-beta-0.001: \dest{não significativo}
      ($p_{adj}=0{,}109$) — embora SRW-l3 tenha maior AP médio em 7 das 8 redes.
      \item PFP-l3 e RA-CNI: estatisticamente indistinguíveis
      ($p_{adj}=0{,}945$).
    \end{itemize}

    \item \textbf{Leitura:} caminhos e passeios curtos oferecem sinais
    preditivos comparáveis; nenhum método domina por larga margem.
    Como o teste usa 8 redes, as conclusões são \textit{indicativas}.
  \end{itemize}

\end{frame}

\section{Discussão e conclusão}

\begin{frame}[t]{Respostas às questões de pesquisa}

  \EspacoAposTitulo

  \begin{itemize}
    \setlength{\itemsep}{\EspacoEntreTopicos}

    \item \textbf{RQ1 — Varia entre domínios?} \dest{Sim.} Desempenho
    alto em redes sociais e de colaboração; baixo em infraestrutura.

    \item \textbf{RQ2 — Quase-locais superam locais simples?}
    \dest{Em agregado, sim}, com vantagem de caminhos/passeios curtos —
    mas não de forma absoluta: RA, AA e CN seguem competitivos.

    \item \textbf{RQ3 — Há método dominante?} SRW-l3 é o melhor geral,
    porém \dest{sem dominância estatística} por larga margem sobre LPI.

    \item \textbf{RQ4 — Estrutura explica a dificuldade?}
    \dest{Exploratoriamente, sim}: maior fechamento local e grau médio
    associam-se a tarefas mais fáceis.
  \end{itemize}

\end{frame}

\begin{frame}[t]{Conclusão}

  \EspacoAposTitulo

  \begin{itemize}
    \setlength{\itemsep}{\EspacoEntreTopicos}

    \item Comparação de \dest{19 métodos topológicos} em \dest{8 redes}
    de 4 domínios, sob protocolo único (5 folds, amostragem balanceada).

    \item \dest{Caminhos e passeios curtos} fornecem informação estrutural
    útil e maior estabilidade entre domínios; \dest{SRW-l3} teve o melhor
    AP médio geral.

    \item Métodos locais simples (RA, AA, CN) permanecem
    \dest{competitivos} e de baixo custo em várias redes.

    \item As propriedades estruturais (grau médio, agrupamento) ajudam a
    explicar a dificuldade da tarefa — relação a ser lida como
    \textit{exploratória}.
  \end{itemize}
